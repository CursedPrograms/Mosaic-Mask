cmake_minimum_required(VERSION 3.20)
project(MosaicMask VERSION 2.0.0 LANGUAGES C CXX)
# Mosaic Mask in C++, for Atelier: your face as a shifting mosaic of strangers, for video calls.
#   MosaicMask   the window (SDL2, OpenGL 3, Dear ImGui): the webcam, the mosaic, the voice, the virtual camera, Record
#   mosaic-cli   one masked frame from a picture (tests, trying settings)
# OpenCV 4.10 (the camera, the warp), ONNX Runtime's C API (the face models, loaded at run time), miniaudio (the voice).
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

set(EXT ${CMAKE_SOURCE_DIR}/external)
set(STB_DIR ${EXT}/stb)
foreach(f stb_image.h stb_image_write.h)
    if(NOT EXISTS ${STB_DIR}/${f})
        file(DOWNLOAD https://raw.githubusercontent.com/nothings/stb/master/${f} ${STB_DIR}/${f})
    endif()
endforeach()
set(ORT_DIR ${EXT}/onnxruntime)
if(NOT EXISTS ${ORT_DIR}/onnxruntime_c_api.h)
    file(DOWNLOAD https://raw.githubusercontent.com/microsoft/onnxruntime/v1.20.1/include/onnxruntime/core/session/onnxruntime_c_api.h ${ORT_DIR}/onnxruntime_c_api.h)
endif()
if(NOT EXISTS ${EXT}/miniaudio.h)
    file(DOWNLOAD https://raw.githubusercontent.com/mackron/miniaudio/0.11.21/miniaudio.h ${EXT}/miniaudio.h)
endif()
if(NOT OpenCV_DIR AND EXISTS "D:/cc/_src/opencv-dl/opencv/build/OpenCVConfig.cmake")
    set(OpenCV_DIR "D:/cc/_src/opencv-dl/opencv/build")
endif()
find_package(OpenCV REQUIRED)

add_library(mosaic_core STATIC src/face.cpp src/mosaic.cpp src/vcam.cpp src/voice.cpp)
target_include_directories(mosaic_core PUBLIC src ${ORT_DIR} ${EXT} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(mosaic_core PUBLIC ${OpenCV_LIBS})
if(MSVC)
    target_compile_options(mosaic_core PRIVATE /O2 /fp:fast /utf-8 /D_CRT_SECURE_NO_WARNINGS)
else()
    target_link_libraries(mosaic_core PUBLIC dl pthread)
endif()

add_executable(mosaic-cli src/cli.cpp)
target_link_libraries(mosaic-cli PRIVATE mosaic_core)
if(MSVC)
    target_compile_options(mosaic-cli PRIVATE /O2 /utf-8 /D_CRT_SECURE_NO_WARNINGS)
endif()

# the window
option(MOSAIC_GUI "Build the window" ON)
if(MOSAIC_GUI)
    include(FetchContent)
    if(EXISTS ${EXT}/imgui-1.90.4/imgui.h)
        set(FETCHCONTENT_SOURCE_DIR_IMGUI ${EXT}/imgui-1.90.4)
    endif()
    FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.90.4.tar.gz)
    FetchContent_MakeAvailable(imgui)
    if(WIN32 AND NOT SDL2_DIR AND EXISTS ${EXT}/SDL2-2.30.9/cmake)
        set(SDL2_DIR ${EXT}/SDL2-2.30.9/cmake)
    endif()
    find_package(SDL2 REQUIRED)
    find_package(OpenGL REQUIRED)
    add_library(imgui STATIC
        ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
        ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
    target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
    target_link_libraries(imgui PUBLIC SDL2::SDL2 OpenGL::GL)
    set(GUI_SOURCES src/gui.cpp src/recorder.cpp src/lang.cpp src/splash.cpp)
    if(WIN32 AND EXISTS ${CMAKE_SOURCE_DIR}/src/app.rc)
        list(APPEND GUI_SOURCES src/app.rc)
    endif()
    add_executable(MosaicMask WIN32 ${GUI_SOURCES})
    target_include_directories(MosaicMask PRIVATE ${EXT} ${STB_DIR})
    target_link_libraries(MosaicMask PRIVATE mosaic_core imgui)
    target_compile_definitions(MosaicMask PRIVATE MOSAIC_VERSION="${PROJECT_VERSION}")
    if(TARGET SDL2::SDL2main)
        target_link_libraries(MosaicMask PRIVATE SDL2::SDL2main)
    endif()
    if(MSVC)
        target_compile_options(MosaicMask PRIVATE /W3 /utf-8 /O2 /D_CRT_SECURE_NO_WARNINGS)
    endif()
    add_custom_command(TARGET MosaicMask POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:MosaicMask>/assets
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/lang $<TARGET_FILE_DIR:MosaicMask>/lang)
    if(WIN32)
        add_custom_command(TARGET MosaicMask POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:MosaicMask>)
    endif()
endif()

# the models, ONNX Runtime and OpenCV next to the exes
add_custom_command(TARGET mosaic-cli POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/models $<TARGET_FILE_DIR:mosaic-cli>/models)
if(WIN32)
    if(EXISTS ${ORT_DIR}/onnxruntime.dll)
        add_custom_command(TARGET mosaic-cli POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ORT_DIR}/onnxruntime.dll $<TARGET_FILE_DIR:mosaic-cli>)
    endif()
    add_custom_command(TARGET mosaic-cli POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${OpenCV_DIR}/x64/vc16/bin/opencv_world4100.dll" "${OpenCV_DIR}/x64/vc16/bin/opencv_videoio_ffmpeg4100_64.dll" $<TARGET_FILE_DIR:mosaic-cli>)
endif()
